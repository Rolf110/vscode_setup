@tool
def current_datetime() -> str:
    """Return the server's current local date/time with timezone."""
    return datetime.now().astimezone().isoformat(timespec="seconds")


TOOLS = [current_datetime]
TOOL_MAP = {t.name: t for t in TOOLS}


for key, default in {

    "regenerate": False,
    "remote_file_ids": [],
}.items():
    if key not in st.session_state:
        st.session_state[key] = default

    effort = None if reasoning_effort == "off" else reasoning_effort

        scope=scope,

        ca_bundle_file=ca_bundle or None,

def attachment_type(mime: str | None) -> str:
    mime = mime or "application/octet-stream"
    if mime.startswith("image/"):
        return "image"
    if mime.startswith("audio/"):
        return "audio"
    return "file"


def upload_attachments(llm: GigaChat, uploaded_files: list[Any]) -> list[dict[str, Any]]:
    result = []
    for f in uploaded_files:
        raw = f.getvalue()
        remote = llm.upload_file((f.name, raw))
        st.session_state.remote_file_ids.append(remote.id_)
        kind = attachment_type(getattr(f, "type", None))
        result.append(
            {
                "name": f.name,
                "mime": getattr(f, "type", None),
                "size": len(raw),
                "kind": kind,
                "file_id": remote.id_,
                # Keep small media locally only for UI preview.
                "preview": raw if kind in {"image", "audio"} and len(raw) <= 10_000_000 else None,
            }
        )
    return result


def make_user_message(text: str, attachments: list[dict[str, Any]]) -> HumanMessage:
    if not attachments:
        return HumanMessage(content=text)

    blocks: list[dict[str, Any]] = []
    if text:
        blocks.append({"type": "text", "text": text})
    blocks.extend({"type": a["kind"], "file_id": a["file_id"]} for a in attachments)
    return HumanMessage(content_blocks=blocks)


def run_llm(llm: GigaChat) -> tuple[Any, list[Any], list[dict[str, Any]]]:
    """Run one complete non-streaming turn, including optional tool calls."""
    messages: list[Any] = []
    if system_prompt.strip():
        messages.append(SystemMessage(content=system_prompt.strip()))
    messages.extend(st.session_state.lc_history)

    start = len(messages)
    runnable = llm.bind_tools(TOOLS, tool_choice="auto") if enable_tools else llm
    tool_events: list[dict[str, Any]] = []

    for _ in range(6):
        reply = runnable.invoke(messages)
        messages.append(reply)

        if not enable_tools or not reply.tool_calls:
            return reply, messages[start:], tool_events

        # GigaChat doesn't support parallel tool calls, but handle defensively.
        for call in reply.tool_calls:
            name = call.get("name", "")
            args = call.get("args") or {}
            tool_impl = TOOL_MAP.get(name)
            try:
                result = tool_impl.invoke(args) if tool_impl else f"Unknown tool: {name}"
            except Exception as e:
                result = f"Tool failed: {type(e).__name__}: {e}"

            tool_events.append({"name": name, "args": args, "result": str(result)})
            messages.append(
                ToolMessage(
                    content=str(result),
                    tool_call_id=call.get("id") or name or "tool",
                )
            )

    raise RuntimeError("Tool-call loop exceeded 6 iterations")


def response_text(reply: Any) -> str:
    if isinstance(reply.content, str):
        return reply.content
    if isinstance(reply.content, list):
        return "\n".join(
            str(x.get("text", "")) if isinstance(x, dict) else str(x)
            for x in reply.content
        )
    return str(reply.content)


def append_assistant(reply: Any, generated: list[Any], tool_events: list[dict[str, Any]]) -> None:
    st.session_state.lc_history.extend(generated)
    meta = dict(getattr(reply, "response_metadata", {}) or {})
    usage = dict(reply.usage_metadata) if getattr(reply, "usage_metadata", None) else None
    st.session_state.ui_history.append(
        {
            "role": "assistant",
            "text": response_text(reply),
            "reasoning": reply.additional_kwargs.get("reasoning_content"),
            "usage": usage,
            "request_id": getattr(reply, "id", None),
            "model": meta.get("model_name", model),
            "finish_reason": meta.get("finish_reason"),
            "tool_events": tool_events,
            "attachments": [],
            "lc_count": len(generated),
        }
    )


def render_chat() -> None:
    for msg in st.session_state.ui_history:
        with st.chat_message(msg["role"]):
            if msg.get("text"):
                st.markdown(msg["text"])

            for a in msg.get("attachments", []):
                if a["kind"] == "image" and a.get("preview") is not None:
                    st.image(a["preview"], caption=a["name"])
                elif a["kind"] == "audio" and a.get("preview") is not None:
                    st.audio(a["preview"], format=a.get("mime") or "audio/wav")
                else:
                    st.caption(f"📎 {a['name']} ({a['size'] / 1024:.1f} KB)")

            if msg["role"] == "assistant":
                if show_reasoning and msg.get("reasoning"):
                    with st.expander("Reasoning"):
                        st.markdown(msg["reasoning"])

                if msg.get("tool_events"):
                    with st.expander("Tool calls"):
                        for event in msg["tool_events"]:
                            st.write(event["name"], event["args"])
                            st.code(event["result"], language=None)

                info = []
                if msg.get("model"):
                    info.append(msg["model"])
                if msg.get("usage"):
                    u = msg["usage"]
                    info.append(
                        f"tokens: {u.get('input_tokens', '?')} in / "
                        f"{u.get('output_tokens', '?')} out / {u.get('total_tokens', '?')} total"
                    )
                if msg.get("finish_reason"):
                    info.append(f"finish: {msg['finish_reason']}")
                if msg.get("request_id"):
                    info.append(f"request: {msg['request_id']}")
                if info:
                    st.caption(" · ".join(info))


def export_json() -> str:
    clean = []
    for msg in st.session_state.ui_history:
        item = {k: v for k, v in msg.items() if k != "lc_count"}
        item["attachments"] = [
            {k: v for k, v in a.items() if k != "preview"}
            for a in item.get("attachments", [])
        ]
        clean.append(item)
    return json.dumps(clean, ensure_ascii=False, indent=2, default=str)

        ca_bundle = st.text_input("CA bundle path", value=os.getenv("GIGACHAT_CA_BUNDLE_FILE", ""))
        timeout = st.number_input("Timeout (sec)", 1, 3600, 180, 10)

        st.session_state.ui_history = []

                try:
                    llm.delete_file(file_id)

    "Message GigaChat…",
    accept_file="multiple",
    accept_audio=True,
    max_upload_size=100,
)

if submission:
    text = (getattr(submission, "text", "") or "").strip()
    files = list(getattr(submission, "files", []) or [])
    if getattr(submission, "audio", None) is not None:
        files.append(submission.audio)

    if text or files:
        try:
            llm = build_llm()
            attachments = []
            if files:
                with st.spinner("Uploading attachments…"):
                    attachments = upload_attachments(llm, files)

            user_msg = make_user_message(text, attachments)
            st.session_state.lc_history.append(user_msg)
            st.session_state.ui_history.append(
                {"role": "user", "text": text, "attachments": attachments}
            )

            with st.spinner("Generating…"):
                reply, generated, events = run_llm(llm)
            append_assistant(reply, generated, events)
            st.rerun()
        except Exception as e:
            st.error(f"{type(e).__name__}: {e}")
