import os, socket, traceback
import torch
import pytorch_lightning as pl


class DeepCrashCallback(pl.Callback):
    """
    Zero-overhead crash dump for multi-GPU/node training.

    REQUIRES this hook on your LightningModule (or LightningDataModule):

        def on_before_batch_transfer(self, batch, dataloader_idx):
            self._cpu_batch = batch
            return batch

    Without it, batches are unrecoverable after sticky CUDA errors
    (illegal memory access, device-side assert).
    """

    def __init__(self, save_dir="crash_dumps", save_weights=False):
        self.save_dir = save_dir
        self.save_weights = save_weights

    # ---- setup check: fail loudly if the hook is missing -------------

    def setup(self, trainer, pl_module, stage):
        if stage != "fit":
            return
        dm = getattr(trainer, "datamodule", None)
        has_hook = any(
            type(obj).on_before_batch_transfer is not pl.LightningModule.on_before_batch_transfer
            for obj in (pl_module, dm) if obj is not None
        )
        if not has_hook:
            rank_zero_warn = pl.utilities.rank_zero.rank_zero_warn
            rank_zero_warn(
                "[DeepCrashCallback] on_before_batch_transfer is not defined. "
                "Batches will NOT be recoverable after sticky CUDA errors."
            )

    # ---- per-step: reference assignment only, no copies, no syncs ----

    def on_train_batch_start(self, trainer, pl_module, batch, batch_idx):
        pl_module._debug = {
            "batch_idx": batch_idx,
            "batch_ref": batch,
            "loss_ref":  None,
        }

    def on_before_backward(self, trainer, pl_module, loss):
        pl_module._debug["loss_ref"] = loss

    # ---- helpers -----------------------------------------------------

    def _dump_path(self, trainer, prefix="crash"):
        return os.path.join(
            self.save_dir,
            f"{prefix}_step{trainer.global_step}"
            f"_rank{trainer.global_rank}_{socket.gethostname()}.pt",
        )

    @staticmethod
    def _safe_cpu(t, name="tensor"):
        if t is None:
            return None
        try:
            return t.detach().cpu()
        except Exception as e:
            return f"<transfer failed '{name}': {type(e).__name__}: {e}>"

    def _safe_cpu_tree(self, obj, name="root"):
        if isinstance(obj, torch.Tensor):
            return self._safe_cpu(obj, name)
        if isinstance(obj, dict):
            return {k: self._safe_cpu_tree(v, k) for k, v in obj.items()}
        if isinstance(obj, (list, tuple)):
            return type(obj)(self._safe_cpu_tree(v, f"{name}[{i}]") for i, v in enumerate(obj))
        return obj

    def _capture_model_state(self, trainer, pl_module):
        strategy = trainer.strategy.__class__.__name__
        try:
            if "FSDP" in strategy:
                return {
                    "strategy": strategy,
                    "note": "local shard only",
                    "local_shard": {n: self._safe_cpu(p, n) for n, p in pl_module.named_parameters()},
                }
            if "DeepSpeed" in strategy:
                return {"strategy": strategy, "note": "use DeepSpeed checkpoint utils offline"}
            return {k: self._safe_cpu(v, k) for k, v in pl_module.state_dict().items()}
        except Exception as e:
            return f"<model state capture failed: {type(e).__name__}: {e}>"

    # ---- crash handler ------------------------------------------------

    def on_exception(self, trainer, pl_module, exception):
        os.makedirs(self.save_dir, exist_ok=True)
        d = getattr(pl_module, "_debug", {})
        path = self._dump_path(trainer)

        payload = {
            "step":        trainer.global_step,
            "epoch":       trainer.current_epoch,
            "global_rank": trainer.global_rank,
            "local_rank":  trainer.local_rank,
            "hostname":    socket.gethostname(),
            "exception":   f"{type(exception).__name__}: {exception}",
            "traceback":   traceback.format_exc(),
            "batch_idx":   d.get("batch_idx"),
            # reliable — already on host
            "cpu_batch":   getattr(pl_module, "_cpu_batch", None),
        }

        # best-effort — works for Python-level exceptions, fails on sticky CUDA
        payload["gpu_batch"] = self._safe_cpu_tree(d.get("batch_ref"), "batch")
        payload["loss"]      = self._safe_cpu(d.get("loss_ref"), "loss")

        try:
            payload["bad_grad_names"] = [
                n for n, p in pl_module.named_parameters()
                if p.grad is not None
                and isinstance(g := self._safe_cpu(p.grad, n), torch.Tensor)
                and (torch.isnan(g).any() or torch.isinf(g).any())
            ]
        except Exception as e:
            payload["bad_grad_names"] = f"<grad scan failed: {e}>"

        if self.save_weights:
            payload["model_state"] = self._capture_model_state(trainer, pl_module)

        torch.save(payload, path)
        print(f"[CRASH] rank {trainer.global_rank} → {path}", flush=True)

        with open(os.path.join(
            self.save_dir,
            f"CRASH_SENTINEL_step{trainer.global_step}_rank{trainer.global_rank}.flag"
        ), "w") as f:
            f.write(payload["traceback"])
