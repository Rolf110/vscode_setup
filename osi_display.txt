import html
import re
import time
import uuid
from datetime import datetime
from IPython.display import HTML, display


_LEVEL_PATTERNS = [
    (re.compile(r"\b(traceback|exception|fatal|critical|error|failed|failure)\b", re.I), "level-error"),
    (re.compile(r"\b(warn|warning)\b", re.I), "level-warn"),
    (re.compile(r"\b(success|succeeded|complete|completed|done)\b", re.I), "level-success"),
    (re.compile(r"\b(info|starting|running)\b", re.I), "level-info"),
]


def _classify_log_line(line: str) -> str:
    for pattern, css_class in _LEVEL_PATTERNS:
        if pattern.search(line):
            return css_class
    return ""


def _render_log_monitor_html(
    *,
    uid: str,
    lines,
    job_id,
    pod_name,
    total_lines,
    shown_lines,
    updated_at,
    poll_interval,
    height,
    status,
    show_line_numbers,
    max_line_chars,
    error_message=None,
):
    rendered_lines = []

    first_line_number = max(total_lines - shown_lines + 1, 1)

    for i, raw_line in enumerate(lines):
        line = str(raw_line)

        if max_line_chars is not None and len(line) > max_line_chars:
            line = line[:max_line_chars] + " …"

        escaped_line = html.escape(line)
        css_class = _classify_log_line(line)

        if show_line_numbers:
            line_no = first_line_number + i
            rendered_lines.append(
                f"""
                <div class="log-line {css_class}">
                    <span class="line-no">{line_no:>6}</span>
                    <span class="line-text">{escaped_line}</span>
                </div>
                """
            )
        else:
            rendered_lines.append(
                f"""
                <div class="log-line {css_class}">
                    <span class="line-text">{escaped_line}</span>
                </div>
                """
            )

    if not rendered_lines:
        rendered_lines.append('<div class="log-line muted">No logs yet.</div>')

    error_box = ""
    if error_message:
        error_box = f"""
        <div class="fetch-error">
            Last fetch error: {html.escape(str(error_message))}
        </div>
        """

    return f"""
    <style>
        #{uid} {{
            border: 1px solid #30363d;
            border-radius: 10px;
            overflow: hidden;
            background: #0d1117;
            color: #c9d1d9;
            font-family: ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, "Liberation Mono", monospace;
        }}

        #{uid} .log-header {{
            background: #161b22;
            border-bottom: 1px solid #30363d;
            padding: 10px 12px;
            font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif;
            display: flex;
            flex-wrap: wrap;
            gap: 10px 16px;
            align-items: center;
            font-size: 13px;
        }}

        #{uid} .badge {{
            padding: 2px 8px;
            border-radius: 999px;
            background: #238636;
            color: white;
            font-weight: 600;
        }}

        #{uid} .badge.stopped {{
            background: #8b949e;
        }}

        #{uid} .meta {{
            color: #8b949e;
        }}

        #{uid} .log-body {{
            height: {height};
            overflow-y: auto;
            padding: 10px 0;
            line-height: 1.45;
            font-size: 13px;
            white-space: pre-wrap;
            word-break: break-word;
        }}

        #{uid} .log-line {{
            padding: 0 12px;
            display: flex;
            gap: 10px;
        }}

        #{uid} .log-line:hover {{
            background: rgba(255, 255, 255, 0.04);
        }}

        #{uid} .line-no {{
            color: #6e7681;
            user-select: none;
            flex: 0 0 auto;
        }}

        #{uid} .line-text {{
            flex: 1 1 auto;
        }}

        #{uid} .level-error {{
            color: #ff7b72;
            font-weight: 600;
        }}

        #{uid} .level-warn {{
            color: #d29922;
            font-weight: 600;
        }}

        #{uid} .level-success {{
            color: #7ee787;
        }}

        #{uid} .level-info {{
            color: #79c0ff;
        }}

        #{uid} .muted {{
            color: #8b949e;
            font-style: italic;
        }}

        #{uid} .fetch-error {{
            background: #3d1f00;
            color: #ffd8a8;
            padding: 8px 12px;
            border-bottom: 1px solid #6e4f00;
            font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif;
            font-size: 13px;
        }}
    </style>

    <div id="{uid}">
        <div class="log-header">
            <span class="badge {'stopped' if status != 'running' else ''}">
                {html.escape(status)}
            </span>
            <strong>job:</strong> <span>{html.escape(str(job_id))}</span>
            <strong>pod:</strong> <span>{html.escape(str(pod_name))}</span>
            <span class="meta">showing {shown_lines} / {total_lines} lines</span>
            <span class="meta">poll: {poll_interval}s</span>
            <span class="meta">updated: {html.escape(updated_at)}</span>
        </div>

        {error_box}

        <div class="log-body" id="{uid}-body">
            {''.join(rendered_lines)}
        </div>
    </div>

    <script>
        const el = document.getElementById("{uid}-body");
        if (el) {{
            el.scrollTop = el.scrollHeight;
        }}
    </script>
    """


def monitor_osiris_logs(
    job_id,
    *,
    osiris_client=None,
    n_last=120,
    poll_interval=2.0,
    timestamps=False,
    pod_index=0,
    height="520px",
    show_line_numbers=True,
    max_line_chars=None,
    continue_on_fetch_error=True,
):
    """
    Stable Jupyter log monitor for a remote Osiris job.

    Parameters
    ----------
    job_id:
        Remote job id.

    osiris_client:
        Optional client object. If omitted, uses global variable `osiris`.

    n_last:
        Number of latest log lines to show.

    poll_interval:
        Seconds between log fetches.

    timestamps:
        Passed to osiris.logs(..., timestamps=timestamps).

    pod_index:
        Which pod from response["pods"] to display.

    height:
        CSS height of the log console.

    show_line_numbers:
        Show absolute line numbers for the displayed tail.

    max_line_chars:
        Optional maximum characters per line.

    continue_on_fetch_error:
        If True, temporary fetch errors are shown in the monitor and polling continues.
        KeyboardInterrupt is never swallowed.
    """

    if osiris_client is None:
        osiris_client = globals().get("osiris")

    if osiris_client is None:
        raise ValueError(
            "No osiris client found. Pass osiris_client=osiris or define global variable `osiris`."
        )

    if poll_interval <= 0:
        raise ValueError("poll_interval must be greater than 0.")

    if n_last is not None and n_last <= 0:
        raise ValueError("n_last must be positive or None.")

    uid = f"osiris-log-monitor-{uuid.uuid4().hex}"

    last_lines = []
    last_pod_name = f"pod[{pod_index}]"
    last_total_lines = 0
    last_shown_lines = 0
    last_error = None

    handle = display(
        HTML(
            _render_log_monitor_html(
                uid=uid,
                lines=[],
                job_id=job_id,
                pod_name=last_pod_name,
                total_lines=0,
                shown_lines=0,
                updated_at=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                poll_interval=poll_interval,
                height=height,
                status="running",
                show_line_numbers=show_line_numbers,
                max_line_chars=max_line_chars,
            )
        ),
        display_id=True,
    )

    try:
        while True:
            try:
                payload = osiris_client.logs(job_id, timestamps=timestamps)
                pods = payload.get("pods", [])

                if not pods:
                    logs = []
                    pod_name = "no pods yet"
                else:
                    if pod_index >= len(pods):
                        raise IndexError(
                            f"pod_index={pod_index} but only {len(pods)} pod(s) available."
                        )

                    pod = pods[pod_index]
                    logs = pod.get("logs", []) or []
                    pod_name = (
                        pod.get("name")
                        or pod.get("pod_name")
                        or pod.get("id")
                        or f"pod[{pod_index}]"
                    )

                total_lines = len(logs)
                shown = logs[-n_last:] if n_last is not None else logs

                last_lines = shown
                last_pod_name = pod_name
                last_total_lines = total_lines
                last_shown_lines = len(shown)
                last_error = None

            except Exception as exc:
                if not continue_on_fetch_error:
                    raise

                last_error = exc

            handle.update(
                HTML(
                    _render_log_monitor_html(
                        uid=uid,
                        lines=last_lines,
                        job_id=job_id,
                        pod_name=last_pod_name,
                        total_lines=last_total_lines,
                        shown_lines=last_shown_lines,
                        updated_at=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                        poll_interval=poll_interval,
                        height=height,
                        status="running",
                        show_line_numbers=show_line_numbers,
                        max_line_chars=max_line_chars,
                        error_message=last_error,
                    )
                )
            )

            time.sleep(poll_interval)

    except KeyboardInterrupt:
        handle.update(
            HTML(
                _render_log_monitor_html(
                    uid=uid,
                    lines=last_lines,
                    job_id=job_id,
                    pod_name=last_pod_name,
                    total_lines=last_total_lines,
                    shown_lines=last_shown_lines,
                    updated_at=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                    poll_interval=poll_interval,
                    height=height,
                    status="stopped",
                    show_line_numbers=show_line_numbers,
                    max_line_chars=max_line_chars,
                    error_message="Stopped by KeyboardInterrupt.",
                )
            )
        )

        # Important: re-raise so the notebook cell stops with KeyboardInterrupt.
        raise
