"""
Join daily transaction aggregates (main source) with credit payment events
(additional source).  Part 1.

Both sources are ``dict[str, np.ndarray]``.

The additional source is at *daily* granularity, so the transaction side is
first aggregated to (agr_sid, day) before matching.  The output keeps every
original main-data key untouched and adds the requested credit/payment fields
as new arrays of the *same length as main data*, filled with NaN/NaT where no
match was found.
"""

from collections import defaultdict

import numpy as np


# --------------------------------------------------------------------------
# Schema of the additional source
# --------------------------------------------------------------------------

# One value per credit (per row of the additional dict).
_CREDIT_FIELDS = (
    "agr_cred_sid",
    "cred_begin_dt",
    "cred_plan_close_dt",
    "cred_fact_close_dt",
    "cred_initial_sum",
    "intrst_rate",
    "credit_type_code",
    "cred_type_code",
)

# Fields stored inside every dict of ``fct_payment_arrays`` (one value per
# payment event of a given credit).
_PAYMENT_FIELDS = (
    "payment_dt",
    "payment_total_amt",
    "payment_primary_amt",
    "payment_interest_amt",
    "overdue_length",
)


# --------------------------------------------------------------------------
# Small helpers
# --------------------------------------------------------------------------

def _to_day(value):
    """Normalise any date-like value to day-precision datetime64."""
    if value is None:
        return np.datetime64("NaT", "D")
    try:
        return np.datetime64(value, "D")
    except Exception:
        return np.datetime64("NaT", "D")


def _classify(sample):
    """Pick an output category for a representative (non-missing) value."""
    if isinstance(sample, np.datetime64) or hasattr(sample, "year"):  # date/datetime
        return "datetime"
    if isinstance(sample, (bool, np.bool_)):
        return "object"
    if isinstance(sample, (int, float, np.integer, np.floating)):
        return "numeric"
    return "object"


def _empty_output(n, kind):
    if kind == "numeric":
        return np.full(n, np.nan, dtype=np.float64)
    if kind == "datetime":
        return np.full(n, np.datetime64("NaT"), dtype="datetime64[us]")
    return np.full(n, np.nan, dtype=object)  # strings / mixed -> object holds NaN


def _coerce(value, kind):
    if value is None:
        return np.nan if kind != "datetime" else np.datetime64("NaT")
    if kind == "numeric":
        try:
            return float(value)
        except (TypeError, ValueError):
            return np.nan
    if kind == "datetime":
        try:
            return np.datetime64(value, "us")
        except Exception:
            return np.datetime64("NaT")
    return value


def _sample_credit_value(add_data, key, indexes):
    arr = add_data[key]
    for ci in indexes:
        v = arr[ci]
        if v is not None and not (isinstance(v, float) and np.isnan(v)):
            return v
    return None


def _sample_payment_value(pay_arrays, key, indexes):
    for ci in indexes:
        rec = pay_arrays[ci]
        if not rec or key not in rec:
            continue
        vals = np.atleast_1d(rec[key])
        if vals.size and vals[0] is not None:
            return vals[0]
    return None


# --------------------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------------------

def join_credit_payments(
    main_data,
    add_data,
    new_keys=None,
    threshold=1.0,
    *,
    card_type="CC",
    cred_type=3,
    # column names -- override if your dict uses different names
    card_col="cust_trans_card_type_code",
    main_agr_col="cust_trans_agr_sid",
    main_date_col="cust_trans_date",
    main_amt_col="cust_trans_rub_amt",
    add_agr_col="agr_cred_sid",
    cred_type_col="cred_type_code",
    payment_arr_col="fct_payment_arrays",
):
    """
    Parameters
    ----------
    main_data, add_data : dict[str, np.ndarray]
    new_keys : list[str] | None
        Which additional-source fields to attach. ``None`` -> every available
        credit field + every payment field.
    threshold : float
        Max allowed |daily_sum - payment_total_amt| for a match.

    Returns
    -------
    dict[str, np.ndarray]
        A copy of ``main_data`` with the new keys added. New arrays have the
        same length as main data; unmatched rows are NaN / NaT.
    """
    n = len(main_data[main_agr_col])

    # ---- 0. resolve which new keys to attach -----------------------------
    credit_fields = [k for k in _CREDIT_FIELDS if k in add_data]
    payment_fields = list(_PAYMENT_FIELDS)
    available = credit_fields + payment_fields
    if new_keys is None:
        new_keys = available
    else:
        unknown = [k for k in new_keys if k not in available]
        if unknown:
            raise KeyError(f"Unknown new_keys {unknown}; available: {available}")

    # ---- 1. daily aggregation of filtered transactions -------------------
    #   filter: card_type == "CC"  AND  amount > 0
    #   group:  (agr_sid, day) -> sum(|amount|)
    card = np.asarray(main_data[card_col])
    amt = np.asarray(main_data[main_amt_col], dtype=np.float64)
    main_mask = (card == card_type) & (amt > 0)

    agr_main = np.asarray(main_data[main_agr_col]).astype(str)
    day_main = np.asarray(main_data[main_date_col]).astype("datetime64[D]")

    groups = {}  # (agr, day) -> {"sum": float, "rows": [row_idx, ...]}
    for i in np.nonzero(main_mask)[0]:
        key = (agr_main[i], day_main[i])
        g = groups.get(key)
        if g is None:
            g = groups[key] = {"sum": 0.0, "rows": []}
        g["sum"] += abs(amt[i])
        g["rows"].append(int(i))

    # ---- 2. filtered payment events from the additional source -----------
    #   filter: cred_type_code == 3, keep those credits' fct_payment_arrays
    cred_codes = np.asarray(add_data[cred_type_col])
    cred_idx = np.nonzero(cred_codes == cred_type)[0]

    agr_add = np.asarray(add_data[add_agr_col]).astype(str)
    pay_arrays = add_data[payment_arr_col]

    # (agr, day) -> list of (credit_index, payment_index, total_amt)
    pay_lookup = defaultdict(list)
    for ci in cred_idx:
        rec = pay_arrays[ci]
        if not rec:
            continue
        dts = np.atleast_1d(rec["payment_dt"])
        totals = np.atleast_1d(rec["payment_total_amt"]).astype(np.float64)
        a = agr_add[ci]
        for j in range(len(dts)):
            pay_lookup[(a, _to_day(dts[j]))].append((int(ci), j, float(totals[j])))

    # ---- 3. allocate output arrays (dtype inferred from a sample) --------
    out, kinds = {}, {}
    for k in new_keys:
        if k in credit_fields:
            sample = _sample_credit_value(add_data, k, cred_idx)
        else:
            sample = _sample_payment_value(pay_arrays, k, cred_idx)
        kinds[k] = _classify(sample)
        out[k] = _empty_output(n, kinds[k])

    # ---- 4. match each daily group to a payment and write back -----------
    for (a, day), g in groups.items():
        candidates = pay_lookup.get((a, day))
        if not candidates:
            continue
        s = g["sum"]
        best, best_diff = None, None
        for ci, j, total in candidates:           # amount condition
            diff = abs(s - total)
            if diff <= threshold and (best_diff is None or diff < best_diff):
                best, best_diff = (ci, j), diff
        if best is None:
            continue

        ci, j = best
        rows = g["rows"]
        for k in new_keys:
            if k in credit_fields:
                val = add_data[k][ci]
            else:
                val = np.atleast_1d(pay_arrays[ci][k])[j]
            out[k][rows] = _coerce(val, kinds[k])

    # ---- 5. assemble result ----------------------------------------------
    result = dict(main_data)        # keep every original key untouched
    result.update(out)
    return result
